90template <class ParamType>
91inline bool loadROSParam(rclcpp::Node::SharedPtr nh, std::string paramName,
92 ParamType &varName) {
93 paramName = normalizeParamName(paramName);
94 rclcpp::Parameter param;
95 if (!nh->get_parameter(paramName, param)) {
96 RCLCPP_ERROR(nh->get_logger(), "Can't find param %s from parameter server", paramName.c_str());
97 return false;
98 }
99 // ROS2 参数严格按类型：整数 override（如 update_rate: 100）会被声明为
100 // PARAMETER_INTEGER，直接按 double 读取会抛 InvalidParameterTypeException。
101 // 这里还原 ROS1 的 int<->double 宽松强转语义。
102 if constexpr (std::is_same_v<ParamType, double> ||
103 std::is_same_v<ParamType, float>) {
104 varName = (param.get_type() == rclcpp::PARAMETER_INTEGER)
105 ? static_cast<ParamType>(param.as_int())
106 : static_cast<ParamType>(param.as_double());
107 } else if constexpr (std::is_same_v<ParamType, int> ||
108 std::is_same_v<ParamType, int64_t>) {
109 varName = (param.get_type() == rclcpp::PARAMETER_DOUBLE)
110 ? static_cast<ParamType>(param.as_double())
111 : static_cast<ParamType>(param.as_int());
112 } else {
113 // 其余类型（string / bool / 各类 vector）要求声明类型严格匹配
114 if (!nh->get_parameter(paramName, varName)) {
115 RCLCPP_ERROR(nh->get_logger(), "Can't find param %s from parameter server", paramName.c_str());
116 return false;
117 }
118 }
119 return true;
120}

131template <class ParamType>
132inline bool loadROSParamDefault(rclcpp::Node::SharedPtr nh, std::string paramName,
133 ParamType &varName, ParamType defaultVal) {
134 paramName = normalizeParamName(paramName);
135 rclcpp::Parameter param;
136 if (!nh->get_parameter(paramName, param)) {
137 varName = defaultVal;
138 RCLCPP_INFO(nh->get_logger(), "Can't find param %s on rosparam server, loading default value.",
139 paramName.c_str());
140 return false;
141 }
142 // 与 loadROSParam 相同的 int<->double 宽容转换
143 if constexpr (std::is_same_v<ParamType, double> ||
144 std::is_same_v<ParamType, float>) {
145 varName = (param.get_type() == rclcpp::PARAMETER_INTEGER)
146 ? static_cast<ParamType>(param.as_int())
147 : static_cast<ParamType>(param.as_double());
148 } else if constexpr (std::is_same_v<ParamType, int> ||
149 std::is_same_v<ParamType, int64_t>) {
150 varName = (param.get_type() == rclcpp::PARAMETER_DOUBLE)
151 ? static_cast<ParamType>(param.as_double())
152 : static_cast<ParamType>(param.as_int());
153 } else {
154 if (!nh->get_parameter(paramName, varName)) {
155 varName = defaultVal;
156 RCLCPP_INFO(nh->get_logger(), "Can't find param %s on rosparam server, loading default value.",
157 paramName.c_str());
158 return false;
159 }
160 }
161 return true;
162}
