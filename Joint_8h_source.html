417 Joint (
418 const JointType joint_type,
419 const Math::Vector3d &joint_axis
420 )
421 {
422 mDoFCount = 1;
423 mJointAxes = new Math::SpatialVector[mDoFCount];
424
425 // Some assertions, as we concentrate on simple cases
426
427 // Only rotation around the Z-axis
428 assert ( joint_type == JointTypeRevolute || joint_type == JointTypePrismatic );
429
430 mJointType = joint_type;
431
432 if (joint_type == JointTypeRevolute) {
433 // make sure we have a unit axis
434 mJointAxes[0].set (
435 joint_axis[0],
436 joint_axis[1],
437 joint_axis[2],
438 0., 0., 0.
439 );
440
441 } else if (joint_type == JointTypePrismatic) {
442 // make sure we have a unit axis
443 assert (joint_axis.squaredNorm() == 1.);
444
445 mJointAxes[0].set (
446 0., 0., 0.,
447 joint_axis[0],
448 joint_axis[1],
449 joint_axis[2]
450 );
451 }
452 }

461 Joint (
462 const Math::SpatialVector &axis_0
463 )
464 {
465 mDoFCount = 1;
466 mJointAxes = new Math::SpatialVector[mDoFCount];
467 mJointAxes[0] = Math::SpatialVector (axis_0);
468 if (axis_0 == Math::SpatialVector(1., 0., 0., 0., 0., 0.)) {
469 mJointType = JointTypeRevoluteX;
470 } else if (axis_0 == Math::SpatialVector(0., 1., 0., 0., 0., 0.)) {
471 mJointType = JointTypeRevoluteY;
472 } else if (axis_0 == Math::SpatialVector(0., 0., 1., 0., 0., 0.)) {
473 mJointType = JointTypeRevoluteZ;
474 } else if (axis_0[0] == 0 &&
475 axis_0[1] == 0 &&
476 axis_0[2] == 0) {
477 mJointType = JointTypePrismatic;
478 } else {
479 mJointType = JointTypeHelical;
480 }
481 validate_spatial_axis (mJointAxes[0]);
482 }

663 bool validate_spatial_axis (Math::SpatialVector &axis)
664 {
665
666 bool axis_rotational = false;
667 bool axis_translational = false;
668
669 Math::Vector3d rotation (axis[0], axis[1], axis[2]);
670 Math::Vector3d translation (axis[3], axis[4], axis[5]);
671
672 if (fabs(rotation.norm()) > 1.0e-8) {
673 axis_rotational = true;
674 }
675
676 if (fabs(translation.norm()) > 1.0e-8) {
677 axis_translational = true;
678 }
679
680 if(axis_rotational && rotation.norm() - 1.0 > 1.0e-8) {
681 std::cerr << "Warning: joint rotation axis is not unit!" << std::endl;
682 }
683
684 if(axis_translational && translation.norm() - 1.0 > 1.0e-8) {
685 std::cerr << "Warning: joint translation axis is not unit!" << std::endl;
686 }
687
688 return axis_rotational != axis_translational;
689 }
