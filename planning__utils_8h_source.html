112 void loadEigenVectorsFromParams() {
113 // Load the gravity vector
114 g_vec << 0, 0, -g;
115
116 // Load the reachability test points
117 reachability_points_body << 0.5 * robot_l, -0.5 * robot_l, 0.5 * robot_l,
118 -0.5 * robot_l, 0.5 * robot_w, 0.5 * robot_w, -0.5 * robot_w,
119 -0.5 * robot_w, 0, 0, 0, 0;
120 // Load the collision test points
121 collision_points_body << 0.5 * robot_l, -0.5 * robot_l, 0.5 * robot_l,
122 -0.5 * robot_l, 0, 0.5 * robot_w, 0.5 * robot_w, -0.5 * robot_w,
123 -0.5 * robot_w, 0, -0.5 * robot_h, -0.5 * robot_h, -0.5 * robot_h,
124 -0.5 * robot_h, -0.5 * robot_h;
125 }

203struct State {
204 Eigen::Vector3d pos;
205 Eigen::Vector3d vel;
206
207 bool operator==(const State rhs) const {
208 // Z velocity is overridden by action
209 return ((pos == rhs.pos) && (vel.head<2>() == rhs.vel.head<2>()));
210 }
211
212 bool operator!=(const State rhs) const {
213 // Z velocity is overridden by action
214 return ((pos != rhs.pos) || (vel.head<2>() != rhs.vel.head<2>()));
215 }
216
217 bool isApprox(const State rhs) const {
218 // Z velocity is overridden by action
219 return ((pos.isApprox(rhs.pos)) &&
220 (vel.head<2>().isApprox(rhs.vel.head<2>())));
221 }
222};
