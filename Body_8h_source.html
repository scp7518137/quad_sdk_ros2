111 void Join (const Math::SpatialTransform &transform, const Body &other_body)
112 {
113 // nothing to do if we join a massles body to the current.
114 if (other_body.mMass == 0. && other_body.mInertia == Math::Matrix3d::Zero()) {
115 return;
116 }
117
118 double other_mass = other_body.mMass;
119 double new_mass = mMass + other_mass;
120
121 if (new_mass == 0.) {
122 throw Errors::RBDLError("Error: cannot join bodies as both have zero mass!\n");
123 }
124
125 Math::Vector3d other_com = transform.E.transpose() * other_body.mCenterOfMass +
126 transform.r;
127 Math::Vector3d new_com = (1 / new_mass ) * (mMass * mCenterOfMass + other_mass *
128 other_com);
129
130 LOG << "other_com = " << std::endl << other_com.transpose() << std::endl;
131 LOG << "rotation = " << std::endl << transform.E << std::endl;
132
133 // We have to transform the inertia of other_body to the new COM. This
134 // is done in 4 steps:
135 //
136 // 1. Transform the inertia from other origin to other COM
137 // 2. Rotate the inertia that it is aligned to the frame of this body
138 // 3. Transform inertia of other_body to the origin of the frame of
139 // this body
140 // 4. Sum the two inertias
141 // 5. Transform the summed inertia to the new COM
142
143 Math::SpatialRigidBodyInertia other_rbi =
144 Math::SpatialRigidBodyInertia::createFromMassComInertiaC (other_body.mMass,
145 other_body.mCenterOfMass, other_body.mInertia);
146 Math::SpatialRigidBodyInertia this_rbi =
147 Math::SpatialRigidBodyInertia::createFromMassComInertiaC (mMass, mCenterOfMass,
148 mInertia);
149
150 Math::Matrix3d inertia_other = other_rbi.toMatrix().block<3,3>(0,0);
151 LOG << "inertia_other = " << std::endl << inertia_other << std::endl;
152
153 // 1. Transform the inertia from other origin to other COM
154 Math::Matrix3d other_com_cross = Math::VectorCrossMatrix(
155 other_body.mCenterOfMass);
156 Math::Matrix3d inertia_other_com = inertia_other - other_mass * other_com_cross
157 * other_com_cross.transpose();
158 LOG << "inertia_other_com = " << std::endl << inertia_other_com << std::endl;
159
160 // 2. Rotate the inertia that it is aligned to the frame of this body
161 Math::Matrix3d inertia_other_com_rotated = transform.E.transpose() *
162 inertia_other_com * transform.E;
163 LOG << "inertia_other_com_rotated = " << std::endl << inertia_other_com_rotated
164 << std::endl;
165
166 // 3. Transform inertia of other_body to the origin of the frame of this body
167 Math::Matrix3d inertia_other_com_rotated_this_origin = Math::parallel_axis (
168 inertia_other_com_rotated, other_mass, other_com);
169 LOG << "inertia_other_com_rotated_this_origin = " << std::endl <<
170 inertia_other_com_rotated_this_origin << std::endl;
171
172 // 4. Sum the two inertias
173 Math::Matrix3d inertia_summed = Math::Matrix3d (this_rbi.toMatrix().block<3,3>
174 (0,0)) + inertia_other_com_rotated_this_origin;
175 LOG << "inertia_summed = " << std::endl << inertia_summed << std::endl;
176
177 // 5. Transform the summed inertia to the new COM
178 Math::Matrix3d new_inertia = inertia_summed - new_mass *
179 Math::VectorCrossMatrix (new_com) * Math::VectorCrossMatrix(
180 new_com).transpose();
181
182 LOG << "new_mass = " << new_mass << std::endl;
183 LOG << "new_com = " << new_com.transpose() << std::endl;
184 LOG << "new_inertia = " << std::endl << new_inertia << std::endl;
185
186 *this = Body (new_mass, new_com, new_inertia);
187 }
