346 unsigned int GetBodyId (const char *body_name) const
347 {
348 if (mBodyNameMap.count(body_name) == 0) {

356 std::string GetBodyName (unsigned int body_id) const
357 {
358 std::map<std::string, unsigned int>::const_iterator iter

360
361 while (iter != mBodyNameMap.end()) {

405 unsigned int GetParentBodyId (unsigned int id)
406 {
407 if (id >= fixed_body_discriminator) {
408 return mFixedBodies[id - fixed_body_discriminator].mMovableParent;
409 }
410
411 unsigned int parent_id = lambda[id];
412
413 while (mBodies[parent_id].mIsVirtual) {
414 parent_id = lambda[parent_id];
415 }
416
417 return parent_id;
418 }

423 Math::SpatialTransform GetJointFrame (unsigned int id)
424 {
425 if (id >= fixed_body_discriminator) {
426 return mFixedBodies[id - fixed_body_discriminator].mParentTransform;
427 }
428
429 unsigned int child_id = id;
430 unsigned int parent_id = lambda[id];
431 if (mBodies[parent_id].mIsVirtual) {
432 while (mBodies[parent_id].mIsVirtual) {
433 child_id = parent_id;
434 parent_id = lambda[child_id];
435 }
436 return X_T[child_id];
437 } else {
438 return X_T[id];
439 }
440 }

445 void SetJointFrame (unsigned int id,
446 const Math::SpatialTransform &transform)
447 {
448 if (id >= fixed_body_discriminator) {
449 throw Errors::RBDLError("Error: setting of parent transform not supported for fixed bodies!");
450 }
451
452 unsigned int child_id = id;
453 unsigned int parent_id = lambda[id];
454 if (mBodies[parent_id].mIsVirtual) {
455 while (mBodies[parent_id].mIsVirtual) {
456 child_id = parent_id;
457 parent_id = lambda[child_id];
458 }
459 X_T[child_id] = transform;
460 } else if (id > 0) {
461 X_T[id] = transform;
462 }
463 }
